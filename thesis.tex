    \vspace{0pt}
    \begin{advantagebox}{#1}
        \begin{itemize}[leftmargin=2.5mm]
            \setlength\itemsep{0.0em}
            #2
        \end{itemize}
    \end{advantagebox}
\end{minipage}
\hfill
\begin{minipage}[t]{0.49\textwidth}
    \vspace{0pt}
    \begin{disadvantagebox}{#1}
        \begin{itemize}[leftmargin=2.5mm]
            \setlength\itemsep{0.0em}
            #3
        \end{itemize}
    \end{disadvantagebox}
\end{minipage}\newline
}

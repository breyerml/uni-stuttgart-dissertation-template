        \Function{update\_accelerations}{\textit{body}, \textit{node}}
            \State calculate direction vector $\vec{d}$ of \textit{body} and the center of mass of \textit{node}
            \If{pseudo-body can be used $\vee$ \textit{node} is leaf}
                \State update acceleration of $\vec{a}_{\textit{body}}$ // TODO: formula?
            \Else
                \ForAll{each \textit{child} of \textit{node}}
                    \State \Call{update\_accelerations}{\textit{body}, \textit{child}}
                \EndFor
            \EndIf
        \EndFunction
    \end{algorithmic}
